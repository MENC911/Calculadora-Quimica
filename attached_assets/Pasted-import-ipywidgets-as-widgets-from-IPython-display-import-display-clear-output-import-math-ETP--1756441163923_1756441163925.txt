    def calcular_mg(output, volver_al_menu):
        with output:
            clear_output()

            # Variables para el cálculo
            parametros = {
                'N_titulante': None,
                'ml_titulante': None,
                'pmeq_muestra': None,
                'aforo': None,
                'alicuota': None,
                'formula': 'sin_aforo'
            }

            # Widgets principales
            N_titulante_input = widgets.FloatText(
                description="Normalidad titulante (N):",
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px')
            )

            ml_titulante_input = widgets.FloatText(
                description="ml gastados de titulante:",
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px')
            )

            pmeq_muestra_input = widgets.FloatText(
                description="pmeq de la muestra:",
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px')
            )

            formula_dropdown = widgets.Dropdown(
                options=[
                    ('Sin aforo/alícuota', 'sin_aforo'),
                    ('Con aforo/alícuota', 'con_aforo')
                ],
                description='Tipo de cálculo:',
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px')
            )

            aforo_input = widgets.FloatText(
                description="Aforo (ml):",
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px'),
                disabled=True
            )

            alicuota_input = widgets.FloatText(
                description="Alícuota (ml):",
                style={'description_width': '200px'},
                layout=widgets.Layout(width='300px'),
                disabled=True
            )

            calcular_btn = widgets.Button(
                description="⚖️ Calcular mg",
                button_style='success',
                layout=widgets.Layout(width='150px')
            )

            limpiar_btn = widgets.Button(
                description="♻ Limpiar",
                button_style='warning',
                layout=widgets.Layout(width='120px')
            )

            volver_btn = widgets.Button(
                description="🏠 Menú",
                button_style='info',
                layout=widgets.Layout(width='120px')
            )

            # Output para resultados
            result_output = widgets.Output(
                layout=widgets.Layout(
                    border='1px solid #eee',
                    padding='10px',
                    margin='10px 0'
                )
            )

            # Función para habilitar/deshabilitar aforo y alícuota
            def toggle_aforo(change):
                aforo_input.disabled = change['new'] == 'sin_aforo'
                alicuota_input.disabled = change['new'] == 'sin_aforo'
                parametros['formula'] = change['new']

            formula_dropdown.observe(toggle_aforo, names='value')

            # Función para calcular los miligramos
            def calcular_miligramos(b):
                with result_output:
                    result_output.clear_output()

                    try:
                        # Validar datos
                        if None in [N_titulante_input.value, ml_titulante_input.value, pmeq_muestra_input.value]:
                            display(widgets.HTML("<div style='color: red;'>❌ Error: Complete todos los campos requeridos</div>"))
                            return

                        if parametros['formula'] == 'con_aforo' and (aforo_input.value is None or alicuota_input.value is None):
                            display(widgets.HTML("<div style='color: red;'>❌ Error: Ingrese valores para aforo y alícuota</div>"))
                            return

                        # Realizar cálculo
                        if parametros['formula'] == 'sin_aforo':
                            mg = N_titulante_input.value * ml_titulante_input.value * pmeq_muestra_input.value * 1000
                        else:
                            if alicuota_input.value == 0:
                                display(widgets.HTML("<div style='color: red;'>❌ Error: La alícuota no puede ser cero</div>"))
                                return
                            mg = (N_titulante_input.value * ml_titulante_input.value *
                                pmeq_muestra_input.value * (aforo_input.value / alicuota_input.value) * 1000)

                        # Mostrar resultados con formato profesional
                        display(widgets.HTML(f"""
                        <div style='background: #f8f9fa; padding: 15px; border-radius: 5px;'>
                            <h3 style='color: #1f77b4; margin-top: 0;'>⚖️ Resultado del Cálculo</h3>

                            <div style='background: #f6ffed; padding: 10px; border-radius: 5px; margin-bottom: 10px;'>
                                <p style='font-size: 1.2em;'>
                                    <strong>Miligramos calculados:</strong>
                                    <span style='color: #fa8c16; font-weight: bold;'>{mg:.4f} mg</span>
                                </p>
                            </div>

                            <div style='background: #e6f7ff; padding: 10px; border-radius: 5px; margin-bottom: 10px;'>
                                <h4 style='margin: 0 0 10px 0; color: #1890ff;'>Parámetros Utilizados</h4>
                                <ul style='margin: 0; padding-left: 20px;'>
                                    <li>Normalidad titulante: {N_titulante_input.value} N</li>
                                    <li>ml gastados de titulante: {ml_titulante_input.value} ml</li>
                                    <li>pmeq de la muestra: {pmeq_muestra_input.value}</li>
                                    {f"<li>Aforo: {aforo_input.value} ml</li><li>Alícuota: {alicuota_input.value} ml</li>" if parametros['formula'] == 'con_aforo' else ""}
                                </ul>
                            </div>

                            <div style='background: #fff7e6; padding: 10px; border-radius: 5px;'>
                                <h4 style='margin: 0 0 10px 0; color: #d46b08;'>Fórmula Aplicada</h4>
                                <p style='margin: 0; font-family: monospace;'>
                                    mg = N × ml × pmeq × 1000 {f"× (aforo/alícuota)" if parametros['formula'] == 'con_aforo' else ""}
                                </p>
                            </div>
                        </div>
                        """))

                    except Exception as e:

            def limpiar_formulario(b):

                ml_titulante_input.value = None
                pmeq_muestra_input.value = None
                aforo_input.value = None
                alicuota_input.value = None
                formula_dropdown.value = 'sin_aforo'
                with result_output:
                    result_output.clear_output()

            # Asignar funciones a los botones
            calcular_btn.on_click(calcular_miligramos)
            limpiar_btn.on_click(limpiar_formulario)
            volver_btn.on_click(volver_al_menu)

            # Mostrar interfaz
            display(widgets.VBox([
                widgets.HTML("""
                <div style='background: #1f77b4; color: white; padding: 10px; border-radius: 5px;'>
                    <h2 style='margin: 0;'>🧪 Calculadora de Miligramos</h2>

                            display(widgets.HTML("<div style='color: red;'>❌ Error: Ingrese valores para aforo y alícuota</div>"))
                            return

                        if gramos_muestra_input.value == 0:
                            display(widgets.HTML("<div style='color: red;'>❌ Error: Los gramos de muestra no pueden ser cero</div>"))
                            return


                        if parametros['formula'] == 'sin_aforo':
                            porcentaje = (N_titulante_input.value *
                                        ml_titulante_input.value *
                                        pmeq_muestra_input.value * 100) / gramos_muestra_input.value
                        else:
                            if alicuota_input.value == 0:
                                display(widgets.HTML("<div style='color: red;'>❌ Error: La alícuota no puede ser cero</div>"))
                                return
                            porcentaje = (N_titulante_input.value *
                                        ml_titulante_input.value *
                                        pmeq_muestra_input.value *
                                        (aforo_input.value / alicuota_input.value) * 100) / gramos_muestra_input.value


                        display(widgets.HTML(f"""
                        <div style='background: #f8f9fa; padding: 15px; border-radius: 5px;'>
                            <h3 style='color: #1f77b4; margin-top: 0;'>🧪 Resultado del Cálculo</h3>

                            <div style='background: #f6ffed; padding: 10px; border-radius: 5px; margin-bottom: 10px;'>
                                <p style='font-size: 1.2em;'>
                                    <strong>Porcentaje del analito:</strong>
                                    <span style='color: #fa8c16; font-weight: bold;'>{porcentaje:.4f}%</span>
                                </p>
                            </div>

                            <div style='background: #e6f7ff; padding: 10px; border-radius: 5px; margin-bottom: 10px;'>
                                <h4 style='margin: 0 0 10px 0; color: #1890ff;'>Parámetros Utilizados</h4>
                                <ul style='margin: 0; padding-left: 20px;'>
                                    <li>Normalidad titulante: {N_titulante_input.value} N</li>
                                    <li>ml gastados de titulante: {ml_titulante_input.value} ml</li>
                                    <li>pmeq del analito: {pmeq_muestra_input.value}</li>
                                    <li>Gramos de muestra: {gramos_muestra_input.value} g</li>
                                    {f"<li>Aforo: {aforo_input.value} ml</li><li>Alícuota: {alicuota_input.value} ml</li>" if parametros['formula'] == 'con_aforo' else ""}
                                </ul>
                            </div>

                            <div style='background: #fff7e6; padding: 10px; border-radius: 5px;'>
                                <h4 style='margin: 0 0 10px 0; color: #d46b08;'>Fórmula Aplicada</h4>
                                <p style='margin: 0; font-family: monospace;'>
                                    % = (N × ml × pmeq × {f"(aforo/alícuota) × " if parametros['formula'] == 'con_aforo' else ""}100) / gramos muestra
                                </p>
                            </div>
                        </div>
                        """))

                    except Exception as e:

            def calcular_potencial(b):
                with result_output:
                    clear_output()
                    try:
                        data1 = compuestos[especie_seleccionada_1]
                        data2 = compuestos[especie_seleccionada_2]

#Cambiar para ver el exceso
                        if data1["potencial_estandar"] > data2["potencial_estandar"]:
                            especie_Red, especie_Ox = especie_seleccionada_2, especie_seleccionada_1

                            conc_Ox = concentracion2_input.value

                        num_electrones = abs(data_Red["carga"] - data_Ox["carga"])
                        if num_electrones == 0:
                            num_electrones = 1

                        pmeq_Red = conc_Red * vol_Red
                        pmeq_Ox = conc_Ox * vol_Ox

                        if pmeq_Ox == 0:
                            display(widgets.HTML("<div style='color: red;'>❌ Error: El volumen o concentración de la especie oxidada no puede ser cero</div>"))

                            display(widgets.HTML("<div style='color: red;'>❌ Error: La cantidad de especie reductora debe ser mayor que la oxidante</div>"))
                            return

                        E = (data_Red["potencial_estandar"] -

                        display(widgets.HTML(f"""

                                    <p><strong>Concentración:</strong> {conc_Red} N</p>
                                    <p><strong>Volumen:</strong> {vol_Red} ml</p>
                                    <p><strong>pmeq:</strong> {pmeq_Red:.4f}</p>
                                    <p><strong>Carga:</strong> {data_Red['carga']}</p>
                                </div>

                                <div style='background: #fff1f0; padding: 10px; border-radius: 5px; width: 48%;'>
                                    <h4 style='margin: 0 0 10px 0; color: #f5222d;'>Especie Oxidante</h4>
                                    <p><strong>Compuesto:</strong> {especie_Ox}</p>
                                    <p><strong>Potencial estándar (E°):</strong> {data_Ox['potencial_estandar']} V</p>
                                    <p><strong>Concentración:</strong> {conc_Ox} N</p>
                                    <p><strong>Volumen:</strong> {vol_Ox} ml</p>
                                    <p><strong>pmeq:</strong> {pmeq_Ox:.4f}</p>
                                    <p><strong>Carga:</strong> {data_Ox['carga']}</p>
                                </div>
                            </div>

                            <div style='margin-top: 15px; background: #f6ffed; padding: 10px; border-radius: 5px;'>
                                <h4 style='margin: 0 0 10px 0; color: #52c41a;'>Detalles del Cálculo</h4>
                                <p><strong>Electrones transferidos (n):</strong> {num_electrones}</p>
                                <p><strong>Ecuación:</strong> E = E° - (0.0592/n) * log[(pmeq_Red - pmeq_Ox)/pmeq_Ox]</p>
                                <p><strong>Cálculo:</strong> E = {data_Red['potencial_estandar']} - (0.0592/{num_electrones}) * log[({pmeq_Red:.4f}-{pmeq_Ox:.4f})/{pmeq_Ox:.4f}]</p>

                                </button>

                        <h3 style='color: #1f77b4; margin-top: 0;'>🧪 Resultado del Cálculo</h3>

                            <h4 style='margin: 0 0 10px 0; color: #1890ff;'>Desglose</h4>
                            <ul style='margin: 0; padding-left: 20px;'>
                    """ +

                    """
                            </ul>
                        </div>
                    </div>
                    """))

            def limpiar_todo(b):
                elementos_agregados.clear()
                calcular_btn.disabled = True
                actualizar_tabla()
                with result_output:
                    result_output.clear_output()

            agregar_btn.on_click(agregar_elemento)
            calcular_btn.on_click(calcular_pm)
            limpiar_btn.on_click(limpiar_todo)
            volver_btn.on_click(volver_al_menu)

                widgets.HTML("""
                <div style='background: #1f77b4; color: white; padding: 10px; border-radius: 5px;'>
                    <h2 style='margin: 0;'>🧬 Calculadora de Peso Molecular</h2>
                    <p style='margin: 5px 0 0 0;'>Agregue elementos para calcular el peso molecular total</p>
                </div>
                """),
