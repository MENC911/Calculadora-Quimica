 def calcular_potencial_electrico(output, volver_al_menu):
        with output:
            clear_output()

            especie_seleccionada_1 = None
            especie_seleccionada_2 = None

            concentracion1_input = widgets.FloatText(
                description="Conc. Especie 1 (N):",
                value=1.0,
                disabled=True,
                style={'description_width': '150px'}
            )
            volumen1_input = widgets.FloatText(
                description="Volumen (ml):",
                value=50.0,
                disabled=True,
                style={'description_width': '150px'}
            )
            concentracion2_input = widgets.FloatText(
                description="Conc. Especie 2 (N):",
                value=1.0,
                disabled=True,
                style={'description_width': '150px'}
            )
            volumen2_input = widgets.FloatText(
                description="Volumen (ml):",
                value=50.0,
                disabled=True,
                style={'description_width': '150px'}
            )

            seleccion1_label = widgets.Label(
                value="🟦 Especie 1: Ninguna selección",
                style={'font_weight': 'bold'}
            )
            seleccion2_label = widgets.Label(
                value="🟥 Especie 2: Ninguna selección",
                style={'font_weight': 'bold'}
            )

            calcular_btn = widgets.Button(
                description="⚡ Calcular Potencial",
                button_style='success',
                disabled=True,
                layout=widgets.Layout(width='auto', margin='5px')
            )
            limpiar_btn = widgets.Button(
                description="♻ Limpiar Selección",
                button_style='warning',
                layout=widgets.Layout(width='auto', margin='5px')
            )
            volver_btn = widgets.Button(
                description="🏠 Volver al menú",
                button_style='info',
                layout=widgets.Layout(width='auto', margin='5px')
            )

            result_output = widgets.Output(
                layout=widgets.Layout(
                    border='1px solid #ccc',
                    padding='10px',
                    margin='10px 0'
                )
            )

            def crear_boton_compuesto(compuesto):
                potencial = compuestos[compuesto]['potencial_estandar']
                color = 'lightcoral' if potencial < 0 else 'lightgreen'

                btn = widgets.Button(
                    description=f"{compuesto} (E°: {potencial} V)",
                    layout=widgets.Layout(
                        width='auto',
                        margin='2px',
                        padding='5px'
                    ),
                    style={'button_color': color},
                    tooltip=f"Carga: {compuestos[compuesto]['carga']}"
                )

                def on_click(b):
                    nonlocal especie_seleccionada_1, especie_seleccionada_2
                    if especie_seleccionada_1 is None:
                        especie_seleccionada_1 = compuesto
                        seleccion1_label.value = f"🟦 Especie 1: {compuesto} | E°: {potencial} V | Carga: {compuestos[compuesto]['carga']}"
                        concentracion1_input.disabled = False
                        volumen1_input.disabled = False
                    elif especie_seleccionada_2 is None and compuesto != especie_seleccionada_1:
                        especie_seleccionada_2 = compuesto
                        seleccion2_label.value = f"🟥 Especie 2: {compuesto} | E°: {potencial} V | Carga: {compuestos[compuesto]['carga']}"
                        concentracion2_input.disabled = False
                        volumen2_input.disabled = False
                        calcular_btn.disabled = False

                    with output:
                        clear_output()
                        display(widgets.VBox([
                            widgets.HTML("<h3 style='color: #1f77b4;'>🔌 Calculadora de Potencial Electroquímico</h3>"),
                            widgets.HBox([seleccion1_label, seleccion2_label]),
                            widgets.HTML("<hr style='margin: 10px 0;'>"),
                            widgets.HBox([
                                widgets.VBox([
                                    widgets.Label("Parámetros Especie 1:"),
                                    concentracion1_input,
                                    volumen1_input
                                ]),
                                widgets.VBox([
                                    widgets.Label("Parámetros Especie 2:"),
                                    concentracion2_input,
                                    volumen2_input
                                ])
                            ]),
                            widgets.HTML("<hr style='margin: 10px 0;'>"),
                            widgets.HBox([calcular_btn, limpiar_btn, volver_btn]),
                            widgets.HTML("<h4 style='margin: 10px 0;'>📋 Lista de Compuestos:</h4>"),
                            grid_botones,
                            result_output
                        ]))

                btn.on_click(on_click)
                return btn

            botones = [crear_boton_compuesto(comp) for comp in sorted(compuestos.keys())]
            grid_botones = widgets.GridBox(
                botones,
                layout=widgets.Layout(
                    grid_template_columns='repeat(4, 250px)',
                    grid_gap='8px',
                    max_height='500px',
                    overflow_y='auto',
                    padding='10px',
                    border='1px solid #eee'
                )
            )

            def calcular_potencial(b):
                with result_output:
                    clear_output()
                    try:
                        data1 = compuestos[especie_seleccionada_1]
                        data2 = compuestos[especie_seleccionada_2]

#Cambiar para ver el exceso
                        if data1["potencial_estandar"] > data2["potencial_estandar"]:
                            especie_Red, especie_Ox = especie_seleccionada_2, especie_seleccionada_1
                            data_Red, data_Ox = data2, data1
                            conc_Red = concentracion2_input.value
                            vol_Red = volumen2_input.value
                            conc_Ox = concentracion1_input.value
                            vol_Ox = volumen1_input.value
                        else:
                            especie_Red, especie_Ox = especie_seleccionada_1, especie_seleccionada_2
                            data_Red, data_Ox = data1, data2
                            conc_Red = concentracion1_input.value
                            vol_Red = volumen1_input.value
                            conc_Ox = concentracion2_input.value
                            vol_Ox = volumen2_input.value

                        num_electrones = abs(data_Red["carga"] - data_Ox["carga"])
                        if num_electrones == 0:
                            num_electrones = 1

                        pmeq_Red = conc_Red * vol_Red
                        pmeq_Ox = conc_Ox * vol_Ox

                        if pmeq_Ox == 0:
                            display(widgets.HTML("<div style='color: red;'>❌ Error: El volumen o concentración de la especie oxidada no puede ser cero</div>"))
                            return

                        if (pmeq_Red - pmeq_Ox) <= 0:
                            display(widgets.HTML("<div style='color: red;'>❌ Error: La cantidad de especie reductora debe ser mayor que la oxidante</div>"))
                            return

                        E = (data_Red["potencial_estandar"] -
                            (0.0592/num_electrones) * math.log10((pmeq_Red - pmeq_Ox)/pmeq_Ox))

                        display(widgets.HTML(f"""
                        <div style='background: #f8f9fa; padding: 15px; border-radius: 5px;'>
                            <h3 style='color: #1f77b4; margin-top: 0;'>⚡ Resultados del Cálculo</h3>

                            <div style='display: flex; justify-content: space-between;'>
                                <div style='background: #e6f7ff; padding: 10px; border-radius: 5px; width: 48%;'>
                                    <h4 style='margin: 0 0 10px 0; color: #1890ff;'>Especie Reductora</h4>
                                    <p><strong>Compuesto:</strong> {especie_Red}</p>
                                    <p><strong>Potencial estándar (E°):</strong> {data_Red['potencial_estandar']} V</p>
                                    <p><strong>Concentración:</strong> {conc_Red} N</p>
                                    <p><strong>Volumen:</strong> {vol_Red} ml</p>
                                    <p><strong>pmeq:</strong> {pmeq_Red:.4f}</p>
                                    <p><strong>Carga:</strong> {data_Red['carga']}</p>
                                </div>

                                <div style='background: #fff1f0; padding: 10px; border-radius: 5px; width: 48%;'>
                                    <h4 style='margin: 0 0 10px 0; color: #f5222d;'>Especie Oxidante</h4>
                                    <p><strong>Compuesto:</strong> {especie_Ox}</p>
                                    <p><strong>Potencial estándar (E°):</strong> {data_Ox['potencial_estandar']} V</p>
                                    <p><strong>Concentración:</strong> {conc_Ox} N</p>
                                    <p><strong>Volumen:</strong> {vol_Ox} ml</p>
                                    <p><strong>pmeq:</strong> {pmeq_Ox:.4f}</p>
                                    <p><strong>Carga:</strong> {data_Ox['carga']}</p>
                                </div>
                            </div>

                            <div style='margin-top: 15px; background: #f6ffed; padding: 10px; border-radius: 5px;'>
                                <h4 style='margin: 0 0 10px 0; color: #52c41a;'>Detalles del Cálculo</h4>
                                <p><strong>Electrones transferidos (n):</strong> {num_electrones}</p>
                                <p><strong>Ecuación:</strong> E = E° - (0.0592/n) * log[(pmeq_Red - pmeq_Ox)/pmeq_Ox]</p>
                                <p><strong>Cálculo:</strong> E = {data_Red['potencial_estandar']} - (0.0592/{num_electrones}) * log[({pmeq_Red:.4f}-{pmeq_Ox:.4f})/{pmeq_Ox:.4f}]</p>
                                <h3 style='margin: 10px 0 0 0;'>Potencial calculado: <span style='color: #fa8c16;'>{E:.4f} V</span></h3>
                            </div>
                        </div>
                        """))

                    except Exception as e:
                        display(widgets.HTML(f"<div style='color: red;'>❌ Error en cálculo: {str(e)}</div>"))

            def limpiar_seleccion(b):
                nonlocal especie_seleccionada_1, especie_seleccionada_2
                especie_seleccionada_1 = None
                especie_seleccionada_2 = None
                seleccion1_label.value = "🟦 Especie 1: Ninguna selección"
                seleccion2_label.value = "🟥 Especie 2: Ninguna selección"
                concentracion1_input.value = 1.0
                volumen1_input.value = 50.0
                concentracion2_input.value = 1.0
                volumen2_input.value = 50.0
                concentracion1_input.disabled = True
                volumen1_input.disabled = True
                concentracion2_input.disabled = True
                volumen2_input.disabled = True
                calcular_btn.disabled = True
                with result_output:
                    clear_output()

            calcular_btn.on_click(calcular_potencial)
            limpiar_btn.on_click(limpiar_seleccion)
            volver_btn.on_click(volver_al_menu)

            display(widgets.VBox([
                widgets.HTML("""
                <div style='background: #1f77b4; color: white; padding: 10px; border-radius: 5px;'>
                    <h2 style='margin: 0;'>Calculadora de Potencial Electroquímico</h2>
                    <p style='margin: 5px 0 0 0;'>Seleccione dos compuestos e ingrese concentraciones y volúmenes</p>
                </div>
                """),
                widgets.HBox([seleccion1_label, seleccion2_label]),
                widgets.HTML("<hr style='margin: 10px 0;'>"),
                widgets.HBox([
                    widgets.VBox([
                        widgets.Label("Parámetros Especie 1:"),
                        concentracion1_input,
                        volumen1_input
                    ]),
                    widgets.VBox([
                        widgets.Label("Parámetros Especie 2:"),
                        concentracion2_input,
                        volumen2_input
                    ])
                ]),
                widgets.HTML("<hr style='margin: 10px 0;'>"),
                widgets.HBox([calcular_btn, limpiar_btn, volver_btn]),
                widgets.HTML("""
                <div style='background: #f0f0f0; padding: 5px 10px; border-radius: 5px; margin: 10px 0;'>
                    <h4 style='margin: 0;'>📋 Lista de Compuestos Disponibles</h4>
                </div>
                """),
                grid_botones,
                result_output
            ]))
